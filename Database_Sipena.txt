  id bigint [pk, increment]

Table laporan_bencana {
  id bigint [pk, increment]
  jenis_bencana jenis_bencana [not null]

  sumber_laporan sumber_laporan [not null]

  kecamatan_id bigint [not null, ref: > kecamatan.id]
  lokasi_text text [not null]
  latitude decimal(10,7) [not null]
  longitude decimal(10,7) [not null]
  location_geom geometry [not null, note: 'Point, SRID 4326']
  sumber_koordinat sumber_koordinat [not null, default: 'titik_peta']
  waktu_kejadian timestamp [not null]
  status status_laporan [not null, default: 'baru']
  severity_level smallint
  is_baru boolean [not null, default: true]
  created_at timestamp [not null]
  updated_at timestamp [not null]
  deleted_at timestamp
  indexes {
    (jenis_bencana)
    (kecamatan_id)
    (status)
    (waktu_kejadian)
    (created_at)
  }
}
