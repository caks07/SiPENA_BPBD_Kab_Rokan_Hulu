'use client'

import dynamic from 'next/dynamic'
import { Skeleton } from '@/components/ui/skeleton'
import type { LaporanBencana, JenisBencana, Kecamatan } from '@/lib/types'

const DisasterMap = dynamic(() => import('./disaster-map'), {
  ssr: false,
  loading: () => (
    <div className="h-full w-full flex items-center justify-center bg-muted rounded-lg">
      <div className="space-y-4 text-center">
        <Skeleton className="h-8 w-8 rounded-full mx-auto" />
        <p className="text-sm text-muted-foreground">Memuat peta...</p>
      </div>
    </div>
  ),
})

interface MapWrapperProps {
  reports: LaporanBencana[]
  jenisBencana?: JenisBencana[]
  kecamatan?: Kecamatan[]
  onReportSelect?: (report: LaporanBencana) => void
  selectedReport?: LaporanBencana | null
  onLocationSelect?: (lat: number, lng: number) => void
  selectedLocation?: { lat: number; lng: number } | null
  isLocationPicker?: boolean
  filterStatus?: string[]
  filterJenisBencana?: number[]
}

export function MapWrapper({
  reports,
  onReportSelect,
  selectedReport,
  onLocationSelect,
  selectedLocation,
  isLocationPicker = false,
  filterStatus = [],
  filterJenisBencana = [],
}: MapWrapperProps) {
  return (
    <DisasterMap
      reports={reports}
      onMarkerClick={onReportSelect}
      onLocationSelect={onLocationSelect}
      selectedLocation={selectedLocation}
      isLocationPicker={isLocationPicker}
      filterStatus={filterStatus}
      filterJenisBencana={filterJenisBencana}
    />
  )
}

export default MapWrapper
